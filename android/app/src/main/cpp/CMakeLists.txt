cmake_minimum_required(VERSION 3.22.1)
project(re2thor C)
set(CMAKE_C_STANDARD 11)
get_filename_component(RE2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
add_library(re2thor SHARED native_inventory.c world.c story.c world_actors.c actor_assets.c combat.c combat_audio.c "${RE2_ROOT}/source/effect.c" "${RE2_ROOT}/source/roomfx.c" "${RE2_ROOT}/source/enemy.c" "${RE2_ROOT}/source/plc.c" "${RE2_ROOT}/source/inventory.c" "${RE2_ROOT}/source/script.c")
target_include_directories(re2thor PRIVATE "${RE2_ROOT}/source")
target_compile_options(re2thor PRIVATE -Wall -Wextra -Werror)

target_compile_definitions(re2thor PRIVATE RE2_HEADLESS)
target_link_libraries(re2thor android m)
